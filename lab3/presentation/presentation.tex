    \begin{minted}{bash}

sudo -i
groupadd wireshark
gpasswd -a dastarikov wireshark
chgrp wireshark /usr/bin/dumpcap
chmod 750 /usr/bin/dumpcap
setcap cap_net_raw,cap_net_admin=eip /usr/bin/dumpcap
