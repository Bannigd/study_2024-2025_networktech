\begin{frame}[fragile]
\frametitle{Настройка DHCP в случае IPv4}
  \begin{minted}[fontsize=\footnotesize]{bash}
    dastarikov@dastarikov-gw-01# set interfaces ethernet eth0 address 10.0.0.1/24
    dastarikov@dastarikov-gw-01# set service dhcp-server shared-network-name dastarikov domain-name dastarikov.net
    dastarikov@dastarikov-gw-01# set service dhcp-server shared-network-name dastarikov name-server 10.0.0.1
    dastarikov@dastarikov-gw-01# set service dhcp-server shared-network-name dastarikov subnet 10.0.0.0/24 default-router 10.0.0.1
    dastarikov@dastarikov-gw-01# set service dhcp-server shared-network-name dastarikov subnet 10.0.0.0/24 range hosts start 10.0.0.2
    dastarikov@dastarikov-gw-01# set service dhcp-server shared-network-name dastarikov subnet 10.0.0.0/24 range hosts stop 10.0.0.253
    dastarikov@dastarikov-gw-01# commit
    dastarikov@dastarikov-gw-01# save
    dastarikov@dastarikov-gw-01# exit
  \end{minted}
\end{frame}

\begin{frame}[fragile]
\frametitle{Настройка DHCP в случае IPv6}
    \begin{minted}[fontsize=\footnotesize]{bash}
      dastarikov@dastarikov-gw-01# set service router-advert interface eth1 prefix 2000::/64
      dastarikov@dastarikov-gw-01# set service router-advert interface eth1 other-config-flag
      dastarikov@dastarikov-gw-01# set service dhcpv6-server shared-network-name dastarikov-stateless
      dastarikov@dastarikov-gw-01# set service dhcpv6-server shared-network-name dastarikov-stateless subnet 2000::0/64
      dastarikov@dastarikov-gw-01# set service dhcpv6-server shared-network-name dastarikov-stateless common-options name-server 2000::1
      dastarikov@dastarikov-gw-01# set service dhcpv6-server shared-network-name dastarikov-stateless common-options domain-search dastarikov.net
      dastarikov@dastarikov-gw-01# commit
      dastarikov@dastarikov-gw-01# save
      dastarikov@dastarikov-gw-01# run show configuration
    \end{minted}
\end{frame}

      dastarikov@dastarikov-gw-01# set service router-advert interface eth2 managed-flag
